# Proposed Finance Schema Blocks

These review-only blocks complete the finance portion of the proposed schema. They will be consolidated with the approved core and extension blocks only when the production schema is authorized.

```prisma
model ChargeType { id String @id @db.Uuid code String @unique @db.VarChar(50) name String @db.VarChar(120) active Boolean @default(true) charges Charge[] @@map("charge_types") }
model Charge { id String @id @db.Uuid chargeNumber String @unique @db.VarChar(50) debtorPartyId String @db.Uuid leaseId String? @db.Uuid propertyId String? @db.Uuid rentableSpaceId String? @db.Uuid serviceEngagementId String? @db.Uuid chargeTypeId String @db.Uuid scheduleId String? @db.Uuid businessDate DateTime @db.Date dueDate DateTime @db.Date currency String @db.Char(3) originalAmount Decimal @db.Decimal(20,4) outstandingAmount Decimal @db.Decimal(20,4) status String @db.VarChar(30) chargeType ChargeType @relation(fields:[chargeTypeId],references:[id]) allocations PaymentAllocation[] invoiceLines InvoiceLine[] @@index([debtorPartyId,status,dueDate]) @@map("charges") }
model Invoice { id String @id @db.Uuid invoiceNumber String @unique @db.VarChar(50) debtorPartyId String @db.Uuid issueDate DateTime @db.Date currency String @db.Char(3) status String @db.VarChar(30) lines InvoiceLine[] @@index([debtorPartyId,issueDate]) @@map("invoices") }
model InvoiceLine { id String @id @db.Uuid invoiceId String @db.Uuid chargeId String @db.Uuid displayAmount Decimal @db.Decimal(20,4) invoice Invoice @relation(fields:[invoiceId],references:[id]) charge Charge @relation(fields:[chargeId],references:[id]) @@unique([invoiceId,chargeId]) @@map("invoice_lines") }
model CreditNote { id String @id @db.Uuid creditNoteNumber String @unique @db.VarChar(50) partyId String @db.Uuid currency String @db.Char(3) amount Decimal @db.Decimal(20,4) businessDate DateTime @db.Date status String @db.VarChar(30) @@map("credit_notes") }
model Waiver { id String @id @db.Uuid chargeId String @db.Uuid amount Decimal @db.Decimal(20,4) reason String @db.Text approvalRequestId String? @db.Uuid createdAt DateTime @db.Timestamptz(6) @@map("waivers") }

model PaymentMethod { id String @id @db.Uuid code String @unique @db.VarChar(40) name String @db.VarChar(100) active Boolean @default(true) @@map("payment_methods") }

model Refund { id String @id @db.Uuid refundNumber String @unique @db.VarChar(50) paymentId String? @db.Uuid partyId String @db.Uuid amount Decimal @db.Decimal(20,4) currency String @db.Char(3) status String @db.VarChar(30) approvalRequestId String? @db.Uuid journalEntryId String? @db.Uuid @@map("refunds") }

model FiscalYear { id String @id @db.Uuid companyId String @db.Uuid name String @db.VarChar(30) startsOn DateTime @db.Date endsOn DateTime @db.Date status String @db.VarChar(20) periods AccountingPeriod[] @@unique([companyId,name]) @@map("fiscal_years") }
model AccountingPeriod { id String @id @db.Uuid fiscalYearId String @db.Uuid periodNo Int startsOn DateTime @db.Date endsOn DateTime @db.Date status PeriodStatus fiscalYear FiscalYear @relation(fields:[fiscalYearId],references:[id]) journals JournalEntry[] @@unique([fiscalYearId,periodNo]) @@map("accounting_periods") }
model Account { id String @id @db.Uuid companyId String @db.Uuid code String @db.VarChar(30) name String @db.VarChar(160) accountType String @db.VarChar(30) normalBalance String @db.VarChar(10) fundClass FundClass? currency String? @db.Char(3) active Boolean @default(true) lines JournalLine[] @@unique([companyId,code]) @@map("accounts") }
model JournalEntry { id String @id @db.Uuid journalNumber String @unique @db.VarChar(50) periodId String @db.Uuid businessDate DateTime @db.Date postedAt DateTime? @db.Timestamptz(6) status JournalStatus currency String @db.Char(3) description String @db.VarChar(300) reversalOfId String? @db.Uuid period AccountingPeriod @relation(fields:[periodId],references:[id]) lines JournalLine[] source JournalSourceLink? @@index([periodId,status,businessDate]) @@map("journal_entries") }
model JournalLine { id String @id @db.Uuid journalEntryId String @db.Uuid lineNo Int accountId String @db.Uuid signedAmount Decimal @db.Decimal(20,4) reportingAmount Decimal? @db.Decimal(20,4) exchangeRate Decimal? @db.Decimal(20,10) branchId String? @db.Uuid propertyId String? @db.Uuid rentableSpaceId String? @db.Uuid ownerPartyId String? @db.Uuid tenantPartyId String? @db.Uuid leaseId String? @db.Uuid vendorPartyId String? @db.Uuid serviceEngagementId String? @db.Uuid journal JournalEntry @relation(fields:[journalEntryId],references:[id]) account Account @relation(fields:[accountId],references:[id]) @@unique([journalEntryId,lineNo]) @@map("journal_lines") }
model JournalSourceLink { id String @id @db.Uuid journalEntryId String @unique @db.Uuid paymentId String? @db.Uuid refundId String? @db.Uuid ownerPayoutId String? @db.Uuid expenseId String? @db.Uuid depositTransactionId String? @db.Uuid brokerageDealId String? @db.Uuid masterLeaseChargeId String? @db.Uuid utilityAllocationId String? @db.Uuid chargeAdjustmentId String? @db.Uuid journal JournalEntry @relation(fields:[journalEntryId],references:[id]) @@map("journal_source_links") }

model OwnerContributionRequest { id String @id @db.Uuid ownerPartyId String @db.Uuid propertyId String? @db.Uuid amount Decimal @db.Decimal(20,4) currency String @db.Char(3) status String @db.VarChar(30) dueDate DateTime? @db.Date @@map("owner_contribution_requests") }
```
